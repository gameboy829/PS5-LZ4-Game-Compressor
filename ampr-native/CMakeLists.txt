cmake_minimum_required(VERSION 3.20)
project(ampr_native LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
add_library(lz4 SHARED reference/external/ampr_emu/third_party/lz4/lib/lz4.c third_party/lz4/lz4hc.c)
target_include_directories(lz4 PRIVATE reference/external/ampr_emu/third_party/lz4/lib third_party/lz4)
target_compile_definitions(lz4 PRIVATE LZ4_DLL_EXPORT=1)
add_executable(ampr-native src/main.cpp reference/external/ampr_emu/third_party/lz4/lib/lz4.c third_party/lz4/lz4hc.c)
target_include_directories(ampr-native PRIVATE reference/external/ampr_emu/include reference/external/ampr_emu/third_party/lz4/lib third_party/lz4 src)
if(WIN32)
  target_link_libraries(ampr-native PRIVATE bcrypt)
endif()
if(MINGW)
  target_link_options(ampr-native PRIVATE -municode -static)
endif()
add_executable(ampr-gui WIN32 src/gui.cpp reference/external/ampr_emu/third_party/lz4/lib/lz4.c third_party/lz4/lz4hc.c)
target_include_directories(ampr-gui PRIVATE reference/external/ampr_emu/include reference/external/ampr_emu/third_party/lz4/lib third_party/lz4 src)
target_link_libraries(ampr-gui PRIVATE bcrypt comctl32 ole32 uuid shell32)
if(MINGW)
  target_link_options(ampr-gui PRIVATE -municode -static)
endif()
